\begin{bt}
Cho hai biểu thức
\[
A=\dfrac{20-2\sqrt{x}}{x-25}+\dfrac{3}{\sqrt{x}+5}
\quad\text{và}\quad
B=\dfrac{\sqrt{x}+2}{\sqrt{x}-5},\quad x\ge 0,\ x\ne 25.
\]
\begin{cacy}(1)
\item Tính giá trị của biểu thức $B$ khi $x=49$.
\item Rút gọn biểu thức $A$.
\item Tìm các giá trị của $x$ để $\dfrac{B}{A}=|x-4|$.
\end{cacy}
\end{bt}
