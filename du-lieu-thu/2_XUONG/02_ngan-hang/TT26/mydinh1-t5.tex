\begin{bt}
Cho hai biểu thức:
\[ A = \dfrac{\sqrt{x}}{\sqrt{x}+2} \quad (x \ge 0) \quad \text{và} \quad B = \left( \dfrac{-2\sqrt{x}+10}{x-25} + \dfrac{\sqrt{x}}{\sqrt{x}+5} \right) : \dfrac{\sqrt{x}+2}{\sqrt{x}+5} \quad \text{với } x \ge 0, x \ne 25. \]
\begin{cacy}
\item Khi $x=9$, tính giá trị biểu thức $A$.
\item Chứng minh: $B = \dfrac{\sqrt{x}-2}{\sqrt{x}+2}$.
\item Đặt $P = \dfrac{A}{B}$. Tìm $x$ để $|P| = P$.
\end{cacy}
\loigiai{}
\end{bt}
