\begin{bt}
Cho hai biểu thức $A = \dfrac{\sqrt{x}}{x+1}$ và $B = \dfrac{x-2}{x+2\sqrt{x}} - \dfrac{1}{\sqrt{x}} + \dfrac{1}{\sqrt{x}+2}$ với $x > 0$.
\begin{cacy}
\item Tính giá trị của $A$ khi $x = 9$.
\item Chứng minh $B = \dfrac{\sqrt{x}-2}{\sqrt{x}}$.
\item Tìm giá trị của $x$ để $P = 2AB + \dfrac{4}{x+1}$ đạt giá trị lớn nhất.
\end{cacy}

\end{bt}
