\begin{bt}
Cho biểu thức $A = \dfrac{2\sqrt{x} - 6}{x - 2\sqrt{x}} - \dfrac{2}{2 - \sqrt{x}} + \dfrac{\sqrt{x} - 3}{\sqrt{x}}$ và $B = \dfrac{\sqrt{x} - 2}{\sqrt{x} + 1}$ với $x > 0$, $x \neq 4$.
\begin{cacy}
\item Tính giá trị của $B$ khi $x = 9$.
\item Rút gọn biểu thức $A$.
\item Tìm các số nguyên $x$ để $\sqrt{AB} < \dfrac{2}{3}$.
\end{cacy}
\loigiai{
\begin{cacy}
\item Thay $x = 9\text{ (tm)}$ vào biểu thức $B$, ta có:
\[ B = \dfrac{\sqrt{9} - 2}{\sqrt{9} + 1} = \dfrac{1}{4}. \]
\item Ta có:
\[ A = \dfrac{2\sqrt{x} - 6}{x - 2\sqrt{x}} - \dfrac{2}{2 - \sqrt{x}} + \dfrac{\sqrt{x} - 3}{\sqrt{x}} \]
\[ A = \dfrac{2\sqrt{x} - 6 + 2\sqrt{x} + (\sqrt{x} - 3)(\sqrt{x} - 2)}{\sqrt{x}(\sqrt{x} - 2)} \]
\[ A = \dfrac{x - \sqrt{x}}{\sqrt{x}(\sqrt{x} - 2)} \]
\[ = \dfrac{\sqrt{x}(\sqrt{x} - 1)}{\sqrt{x}(\sqrt{x} - 2)} = \dfrac{\sqrt{x} - 1}{\sqrt{x} - 2}. \]
\item Ta có $\sqrt{AB} = \sqrt{\dfrac{\sqrt{x} - 1}{\sqrt{x} + 1}}$. ĐK: $x \ge 1$, $x \neq 4$.\\
$\sqrt{AB} < \dfrac{2}{3}$ suy ra $\dfrac{\sqrt{x} - 1}{\sqrt{x} + 1} < \dfrac{4}{9}$, tính được $x < \dfrac{169}{25}$.\\
Vì $x$ nguyên, $x \ge 1$, $x \neq 4$ nên $x \in \{1; 2; 3; 5; 6\}$.
\end{cacy}
}
\end{bt}
