\begin{bt}
Cho các biểu thức: $A = \dfrac{\sqrt{x}+4}{\sqrt{x}+1}$ và $B = \left( \dfrac{\sqrt{x}-2}{\sqrt{x}-4} + \dfrac{\sqrt{x}}{\sqrt{x}+4} - \dfrac{x}{x-16} \right) \left( 1 - \dfrac{\sqrt{x}}{\sqrt{x}+2} \right)$ với $x \ge 0, x \ne 16$.
\begin{cacy}
\item Tính giá trị của biểu thức $A$ khi $x = \dfrac{9}{16}$.
\item Chứng minh $B = \dfrac{2}{\sqrt{x}+4}$.
\item Đặt $P = A \cdot B$, với $x > 0, x \ne 16$, chứng minh $\dfrac{P^2}{4}$ không có giá trị là số nguyên.
\end{cacy}
\loigiai{}
\end{bt}
