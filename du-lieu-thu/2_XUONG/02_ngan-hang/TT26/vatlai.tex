\begin{bt}
Cho biểu thức $A = \dfrac{\sqrt{x}-2}{\sqrt{x}-1}$; $B = \dfrac{\sqrt{x}}{\sqrt{x}+1} + \dfrac{1}{1-\sqrt{x}} + \dfrac{2}{x-1}$ với $x \ge 0; x \ne 1$.
\begin{cacy}(1)
\item Tính giá trị biểu thức $A$ tại $x = 9$.
\item Chứng minh $B = \dfrac{\sqrt{x}-1}{\sqrt{x}+1}$.
\item Cho $P = A.B$. Tìm các số nguyên tố $x$ để $|P| + P = 0$.
\end{cacy}

\end{bt}
